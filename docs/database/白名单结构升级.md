# 白名单结构升级（已有数据库）

新版将白名单改为人员主表和角色关联表。`ddl.sql`、`init.sql` 仅用于新库；已有库不能直接重新执行初始化脚本。以下示例适用于升级前的旧版 `reviewer_whitelist` 表。升级前停止所有应用实例，备份数据库并确认备份可恢复；在测试库先验证。请先核对相同工号的姓名、邮箱等资料是否有冲突；以下按工号优先选择最早的有效记录作为人员资料，全部已删除时选择最早记录。原表保留为备份，不会自动删除。

MySQL 的 DDL 会隐式提交，不能依靠事务整体回滚。升级期间不得并发写入白名单；若任一步失败，先保留所有表并核对状态，不要重复执行建表或重命名。升级后的人员 ID 采用每人工号选中的旧记录 ID；同一工号其他旧记录的角色映射 ID 不再作为人员 ID。

```sql
-- 先确认临时表、角色表和备份表不存在；若已存在请停止并人工检查。
CREATE TABLE reviewer_whitelist_v2 (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    employee_no VARCHAR(64) NOT NULL,
    display_name VARCHAR(100),
    email VARCHAR(200),
    mobile VARCHAR(32),
    department_name VARCHAR(100),
    created_by_employee_no VARCHAR(64) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_by_employee_no VARCHAR(64),
    deleted_at TIMESTAMP,
    UNIQUE KEY uk_reviewer_whitelist_employee (employee_no)
);

CREATE TABLE reviewer_whitelist_role (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    employee_no VARCHAR(64) NOT NULL,
    review_role VARCHAR(64) NOT NULL,
    created_by_employee_no VARCHAR(64) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_by_employee_no VARCHAR(64),
    deleted_at TIMESTAMP,
    UNIQUE KEY uk_whitelist_role_employee (employee_no, review_role),
    INDEX idx_whitelist_role_role_employee (review_role, employee_no),
    CONSTRAINT fk_whitelist_role_employee FOREIGN KEY (employee_no) REFERENCES reviewer_whitelist_v2(employee_no)
);

INSERT INTO reviewer_whitelist_v2
    (id, employee_no, display_name, email, mobile, department_name, created_by_employee_no, created_at,
     deleted, deleted_by_employee_no, deleted_at)
SELECT old.id, old.employee_no, old.display_name, old.email, old.mobile, old.department_name,
       old.created_by_employee_no, old.created_at, old.deleted, old.deleted_by_employee_no, old.deleted_at
FROM reviewer_whitelist old
WHERE old.id = (
    SELECT COALESCE(MIN(CASE WHEN candidate.deleted = FALSE THEN candidate.id END), MIN(candidate.id))
    FROM reviewer_whitelist candidate WHERE candidate.employee_no = old.employee_no
);

INSERT INTO reviewer_whitelist_role
    (employee_no, review_role, created_by_employee_no, created_at, deleted, deleted_by_employee_no, deleted_at)
SELECT employee_no, review_role, created_by_employee_no, created_at, deleted, deleted_by_employee_no, deleted_at
FROM reviewer_whitelist;

-- 人员、角色和启用状态核对通过后，才执行原子重命名。
SELECT COUNT(DISTINCT employee_no) AS old_people FROM reviewer_whitelist;
SELECT COUNT(*) AS new_people FROM reviewer_whitelist_v2;
SELECT COUNT(*) AS old_roles FROM reviewer_whitelist;
SELECT COUNT(*) AS new_roles FROM reviewer_whitelist_role;
SELECT COUNT(*) AS old_active_roles FROM reviewer_whitelist WHERE deleted = FALSE;
SELECT COUNT(*) AS new_active_roles FROM reviewer_whitelist_role WHERE deleted = FALSE;

RENAME TABLE reviewer_whitelist TO reviewer_whitelist_before_person_roles,
             reviewer_whitelist_v2 TO reviewer_whitelist;
```

重命名后检查角色表外键是否指向新的 `reviewer_whitelist`，再核对按人分页、按角色查询、删除和恢复行为，最后部署新版应用。升级成功后仍保留旧表备份，按公司备份策略处理。迁移未在本项目中自动执行，也未连接或修改现有数据库。
